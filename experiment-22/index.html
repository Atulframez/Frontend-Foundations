<?php
/**
 * Experiment 22: PHP Script to Implement MySQL Commands:
 * a) DELETE  b) ORDER BY  c) UPDATE
 * Course: Advanced Web Technology (CSIT248) - Amity University Noida
 */

$pdo = new PDO("mysql:host=localhost;dbname=company_db;charset=utf8mb4", "root", "", [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
]);

// ==========================================
// a) DELETE Command Implementation
// ==========================================
if (isset($_GET['action']) && $_GET['action'] === 'delete' && isset($_GET['id'])) {
    $delete_id = (int)$_GET['id'];
    $stmt = $pdo->prepare("DELETE FROM employees WHERE emp_id = :id");
    $stmt->execute([':id' => $delete_id]);
    echo "Employee #$delete_id successfully removed from database.";
}

// ==========================================
// c) UPDATE Command Implementation
// ==========================================
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'update') {
    $id    = (int)$_POST['emp_id'];
    $desig = trim($_POST['designation']);
    $sal   = (float)$_POST['salary'];
    $dept  = trim($_POST['department']);

    $stmt = $pdo->prepare("UPDATE employees SET designation = :desig, salary = :sal, department = :dept WHERE emp_id = :id");
    $stmt->execute([
        ':desig' => $desig,
        ':sal'   => $sal,
        ':dept'  => $dept,
        ':id'    => $id
    ]);
    echo "Employee #$id successfully updated.";
}

// ==========================================
// b) ORDER BY Command Implementation
// ==========================================
$allowedCols = ['emp_id', 'emp_name', 'department', 'designation', 'salary', 'joined_date'];
$sort = in_array($_GET['sort'] ?? '', $allowedCols) ? $_GET['sort'] : 'emp_id';
$order = (strtoupper($_GET['order'] ?? '') === 'DESC') ? 'DESC' : 'ASC';

$stmt = $pdo->query("SELECT * FROM employees ORDER BY $sort $order");
$employees = $stmt->fetchAll();
?>

Experiment 22: Academic Lab Manual

1. Aim

To write and implement a PHP script that interacts with a MySQL database to execute the standard SQL commands: a) DELETE, b) ORDER BY, and c) UPDATE with parameter binding and exception handling.

2. Command Syntax & Theoretical Notes

  • a) DELETE: Removes specific tuples matching criteria:
    DELETE FROM employees WHERE emp_id = ?;
    Security note: Always use a strict WHERE clause and prepared statement to prevent accidental table truncation.
  • b) ORDER BY: Sorts the result set by one or more columns in ascending (default) or descending order:
    SELECT * FROM employees ORDER BY salary DESC;
    Security note: Column names cannot be bound as PDO parameters; always sanitize through an explicit whitelist array.
  • c) UPDATE: Modifies existing attribute values in one or more records:
    UPDATE employees SET designation = ?, salary = ? WHERE emp_id = ?;
