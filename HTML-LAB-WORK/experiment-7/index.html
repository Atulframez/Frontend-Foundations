📖 JavaScript History Class Implementation Reference 

// 1. Move one page backward in browser history
function goBack() {
  window.history.back(); // Equivalent to window.history.go(-1)
}

// 2. Move one page forward in browser history
function goForward() {
  window.history.forward(); // Equivalent to window.history.go(1)
}

// 3. Jump specific relative offset
function jumpHistory(offset) {
  window.history.go(offset);
}

// 4. Inspect session history length
console.log("Total entries in session:", window.history.length);
